/**
 * Expose a seller's profile phone through the public users REST response only
 * when the seller has enabled "Display Phone Number Publicly".
 * Code Snippets: add as a PHP snippet and run everywhere.
 */

add_action( 'rest_api_init', function () {
	register_rest_route( 'partexhub/v1', '/profile/phone', array(
		array(
			'methods'             => WP_REST_Server::READABLE,
			'callback'            => function () {
				return rest_ensure_response( array(
					'phone_number' => (string) get_user_meta( get_current_user_id(), 'phone_number', true ),
				) );
			},
			'permission_callback' => function () {
				return get_current_user_id() > 0;
			},
		),
		array(
			'methods'             => WP_REST_Server::EDITABLE,
			'callback'            => function ( WP_REST_Request $request ) {
				$phone = trim( sanitize_text_field( $request->get_param( 'phone_number' ) ) );

				if ( $phone && ( ! preg_match( '/^[+()\-\s\d.]+$/', $phone ) || strlen( preg_replace( '/\D/', '', $phone ) ) < 7 ) ) {
					return new WP_Error( 'invalid_phone_number', 'Enter a valid phone number.', array( 'status' => 400 ) );
				}

				update_user_meta( get_current_user_id(), 'phone_number', $phone );

				return rest_ensure_response( array(
					'success'      => true,
					'phone_number' => $phone,
				) );
			},
			'permission_callback' => function () {
				return get_current_user_id() > 0;
			},
		),
	) );

	register_rest_field( 'user', 'partexhub_public_phone', array(
		'get_callback' => function ( $user ) {
			$user_id = isset( $user['id'] ) ? absint( $user['id'] ) : 0;

			if ( ! $user_id ) {
				return '';
			}

			$is_public = (string) get_user_meta( $user_id, 'phone_number_public', true );

			if ( ! in_array( $is_public, array( 'yes', '1', 'true' ), true ) ) {
				return '';
			}

			return sanitize_text_field( get_user_meta( $user_id, 'phone_number', true ) );
		},
		'schema' => array(
			'description' => 'Public seller phone number.',
			'type'        => 'string',
			'context'     => array( 'view', 'edit' ),
			'readonly'    => true,
		),
	) );

	register_rest_field( 'user', 'partexhub_phone_display', array(
		'get_callback' => function ( $user ) {
			$user_id = isset( $user['id'] ) ? absint( $user['id'] ) : 0;
			if ( ! $user_id ) return '';

			$phone = sanitize_text_field( get_user_meta( $user_id, 'phone_number', true ) );
			if ( ! $phone ) return '';

			$is_public = (string) get_user_meta( $user_id, 'phone_number_public', true );
			if ( in_array( $is_public, array( 'yes', '1', 'true' ), true ) ) return $phone;

			$digits = preg_replace( '/\D/', '', $phone );
			if ( ! $digits ) return '';
			$visible_digits = substr( $digits, -4 );
			return str_repeat( '*', max( 3, strlen( $digits ) - strlen( $visible_digits ) ) ) . $visible_digits;
		},
		'schema' => array(
			'description' => 'Public or safely masked seller phone number.',
			'type'        => 'string',
			'context'     => array( 'view', 'edit' ),
			'readonly'    => true,
		),
	) );
} );
