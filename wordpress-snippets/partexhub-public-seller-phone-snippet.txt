/**
 * Expose a seller's profile phone through the public users REST response only
 * when the seller has enabled "Display Phone Number Publicly".
 * Code Snippets: add as a PHP snippet and run everywhere.
 */

add_action( 'rest_api_init', function () {
	register_rest_field( 'user', 'partexhub_public_phone', array(
		'get_callback' => function ( $user ) {
			$user_id = isset( $user['id'] ) ? absint( $user['id'] ) : 0;

			if ( ! $user_id ) {
				return '';
			}

			$is_public = (string) get_user_meta( $user_id, 'phone_number_public', true );

			if ( ! in_array( $is_public, array( 'yes', '1', 'true' ), true ) ) {
				return '';
			}

			return sanitize_text_field( get_user_meta( $user_id, 'phone_number', true ) );
		},
		'schema' => array(
			'description' => 'Public seller phone number.',
			'type'        => 'string',
			'context'     => array( 'view', 'edit' ),
			'readonly'    => true,
		),
	) );
} );

/**
 * Keep the existing authenticated PartexHub profile endpoint compatible with
 * the JetEngine phone_number user-meta field.
 */
add_filter( 'rest_request_after_callbacks', function ( $response, $handler, $request ) {
	if ( '/partexhub/v1/profile' !== $request->get_route() || is_wp_error( $response ) ) {
		return $response;
	}

	$user_id = get_current_user_id();
	if ( ! $user_id ) {
		return $response;
	}

	if ( in_array( $request->get_method(), array( 'POST', 'PUT', 'PATCH' ), true ) && null !== $request->get_param( 'phone_number' ) ) {
		update_user_meta( $user_id, 'phone_number', sanitize_text_field( $request->get_param( 'phone_number' ) ) );
	}

	$response = rest_ensure_response( $response );
	$data     = $response->get_data();

	if ( is_array( $data ) ) {
		$phone                = (string) get_user_meta( $user_id, 'phone_number', true );
		$data['phone']        = $phone;
		$data['phone_number'] = $phone;
		$response->set_data( $data );
	}

	return $response;
}, 10, 3 );
