function partexhub_msg_tables() {
    global $wpdb;
    return array(
        'conversations' => $wpdb->prefix . 'jet_cct_conversations',
        'messages'      => $wpdb->prefix . 'jet_cct_messages',
        'reactions'     => $wpdb->prefix . 'jet_cct_message_reactions',
    );
}

function partexhub_msg_permission() {
    return get_current_user_id() > 0
        ? true
        : new WP_Error('partexhub_unauthorized', 'Authentication required.', array('status' => 401));
}

function partexhub_msg_tables_ready() {
    global $wpdb;
    foreach (partexhub_msg_tables() as $table) {
        if ($wpdb->get_var($wpdb->prepare('SHOW TABLES LIKE %s', $wpdb->esc_like($table))) !== $table) {
            return new WP_Error('partexhub_cct_missing', 'Messaging storage is not configured.', array('status' => 503));
        }
    }
    return true;
}

function partexhub_msg_time($value) {
    if (!$value) return null;
    $timestamp = strtotime($value . (preg_match('/(?:Z|[+-]\d\d:\d\d)$/', $value) ? '' : ' UTC'));
    return $timestamp ? gmdate('c', $timestamp) : null;
}

function partexhub_msg_avatar($user_id) {
    $value = get_user_meta($user_id, 'profile_picture', true);
    $media_id = is_numeric($value) ? absint($value) : 0;
    if (is_array($value)) $media_id = absint($value['id'] ?? $value['attachment_id'] ?? 0);
    if ($media_id) {
        $url = wp_get_attachment_image_url($media_id, 'thumbnail');
        if ($url) return esc_url_raw($url);
    }
    if (is_string($value) && filter_var($value, FILTER_VALIDATE_URL)) return esc_url_raw($value);
    return null;
}

function partexhub_msg_conversation_row($row, $current_user_id) {
    $buyer_id = absint($row->buyer_id);
    $seller_id = absint($row->seller_id);
    $other_id = $current_user_id === $buyer_id ? $seller_id : $buyer_id;
    $other = get_userdata($other_id);
    $listing_id = absint($row->listing_id);
    return array(
        'id'              => absint($row->_ID),
        'listingId'       => $listing_id,
        'listingTitle'    => html_entity_decode(get_the_title($listing_id), ENT_QUOTES, get_bloginfo('charset')),
        'participant'     => array(
            'id'          => $other_id,
            'displayName' => $other ? $other->display_name : 'Marketplace user',
            'avatarUrl'   => partexhub_msg_avatar($other_id),
        ),
        'lastMessage'     => (string) ($row->last_message ?? ''),
        'lastMessageTime' => partexhub_msg_time($row->last_message_time ?? ''),
        'unreadCount'     => $current_user_id === $buyer_id ? absint($row->buyer_unread_count) : absint($row->seller_unread_count),
        'status'          => (string) ($row->conversation_status ?: 'active'),
    );
}

function partexhub_msg_get_owned_conversation($conversation_id, $user_id) {
    global $wpdb;
    $table = partexhub_msg_tables()['conversations'];
    $row = $wpdb->get_row($wpdb->prepare("SELECT * FROM {$table} WHERE _ID = %d", $conversation_id));
    if (!$row) return new WP_Error('partexhub_not_found', 'Conversation not found.', array('status' => 404));
    if (absint($row->buyer_id) !== $user_id && absint($row->seller_id) !== $user_id) {
        return new WP_Error('partexhub_forbidden', 'You cannot access this conversation.', array('status' => 403));
    }
    return $row;
}

function partexhub_msg_excerpt($message) {
    $message = trim(wp_strip_all_tags((string) $message));
    return mb_strlen($message) > 140 ? mb_substr($message, 0, 137) . '...' : $message;
}

function partexhub_msg_normalize_messages($rows, $current_user_id) {
    global $wpdb;
    if (!$rows) return array();
    $tables = partexhub_msg_tables(); $message_ids = array(); $reply_ids = array();
    foreach ($rows as $row) { $message_ids[] = absint($row->_ID); if (absint($row->reply_to_message_id ?? 0)) $reply_ids[] = absint($row->reply_to_message_id); }
    $reply_rows = array();
    if ($reply_ids) { $ids = implode(',', array_map('absint', array_unique($reply_ids))); foreach ($wpdb->get_results("SELECT _ID,sender_id,message_text,conversation_id FROM {$tables['messages']} WHERE _ID IN ({$ids})") as $reply) $reply_rows[absint($reply->_ID)] = $reply; }
    $grouped = array();
    $ids = implode(',', array_map('absint', array_unique($message_ids)));
    foreach ($wpdb->get_results("SELECT message_id,user_id,reaction FROM {$tables['reactions']} WHERE message_id IN ({$ids})") as $reaction) {
        $message_id = absint($reaction->message_id); $type = (string) $reaction->reaction;
        if (!isset($grouped[$message_id][$type])) $grouped[$message_id][$type] = array('type'=>$type,'count'=>0,'reactedByCurrentUser'=>false);
        $grouped[$message_id][$type]['count']++; if (absint($reaction->user_id) === $current_user_id) $grouped[$message_id][$type]['reactedByCurrentUser'] = true;
    }
    return array_map(function($row) use ($reply_rows, $grouped) {
        $id = absint($row->_ID); $reply_id = absint($row->reply_to_message_id ?? 0); $reply = $reply_rows[$reply_id] ?? null;
        $reply_data = null;
        if ($reply_id) $reply_data = $reply && absint($reply->conversation_id) === absint($row->conversation_id) ? array('id'=>$reply_id,'senderId'=>absint($reply->sender_id),'message'=>partexhub_msg_excerpt($reply->message_text)) : array('id'=>$reply_id,'senderId'=>0,'message'=>'Original message unavailable');
        return array('id'=>$id,'conversationId'=>absint($row->conversation_id),'senderId'=>absint($row->sender_id),'receiverId'=>absint($row->receiver_id),'messageText'=>(string)$row->message_text,'isRead'=>(bool)$row->is_read,'sentAt'=>partexhub_msg_time($row->sent_at),'messageType'=>(string)($row->message_type ?: 'text'),'replyTo'=>$reply_data,'reactions'=>array_values($grouped[$id] ?? array()));
    }, $rows);
}

function partexhub_get_conversations() {
    global $wpdb;
    $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $table = partexhub_msg_tables()['conversations'];
    $rows = $wpdb->get_results($wpdb->prepare("SELECT * FROM {$table} WHERE buyer_id = %d OR seller_id = %d ORDER BY COALESCE(last_message_time, cct_modified, cct_created) DESC", $user_id, $user_id));
    return rest_ensure_response(array('conversations' => array_map(function($row) use ($user_id) { return partexhub_msg_conversation_row($row, $user_id); }, $rows)));
}

function partexhub_create_conversation(WP_REST_Request $request) {
    global $wpdb;
    $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $listing_id = absint($request['listing_id']); $receiver_id = absint($request['receiver_id']);
    $listing = get_post($listing_id);
    if (!$listing || $listing->post_type !== 'car-listing') return new WP_Error('partexhub_listing_missing', 'Listing not found.', array('status' => 404));
    $seller_id = absint($listing->post_author);
    if (!$seller_id || !$receiver_id || !get_userdata($receiver_id)) return new WP_Error('partexhub_invalid_receiver', 'Receiver not found.', array('status' => 400));
    if ($user_id === $seller_id) { $buyer_id = $receiver_id; }
    else { $buyer_id = $user_id; if ($receiver_id !== $seller_id) return new WP_Error('partexhub_invalid_receiver', 'The receiver must be the listing seller.', array('status' => 400)); }
    if ($buyer_id === $seller_id) return new WP_Error('partexhub_self_message', 'You cannot message yourself.', array('status' => 400));
    $table = partexhub_msg_tables()['conversations'];
    $existing = $wpdb->get_var($wpdb->prepare("SELECT _ID FROM {$table} WHERE listing_id=%d AND buyer_id=%d AND seller_id=%d LIMIT 1", $listing_id, $buyer_id, $seller_id));
    if ($existing) return rest_ensure_response(array('success' => true, 'conversationId' => absint($existing), 'created' => false));
    $now = current_time('mysql', true);
    $ok = $wpdb->insert($table, array('cct_status'=>'publish','cct_author_id'=>$user_id,'cct_created'=>$now,'cct_modified'=>$now,'listing_id'=>$listing_id,'buyer_id'=>$buyer_id,'seller_id'=>$seller_id,'last_message'=>'','last_message_time'=>$now,'buyer_unread_count'=>0,'seller_unread_count'=>0,'conversation_status'=>'active'), array('%s','%d','%s','%s','%d','%d','%d','%s','%s','%d','%d','%s'));
    if (!$ok) return new WP_Error('partexhub_create_failed', 'Conversation could not be created.', array('status' => 500));
    return new WP_REST_Response(array('success'=>true,'conversationId'=>absint($wpdb->insert_id),'created'=>true), 201);
}

function partexhub_get_conversation(WP_REST_Request $request) {
    global $wpdb;
    $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $conversation_id = absint($request['id']);
    $conversation = partexhub_msg_get_owned_conversation($conversation_id, $user_id); if (is_wp_error($conversation)) return $conversation;
    $page = max(1, absint($request->get_param('page') ?: 1)); $per_page = min(100, max(1, absint($request->get_param('per_page') ?: 50))); $offset = ($page - 1) * $per_page;
    $table = partexhub_msg_tables()['messages'];
    $total = absint($wpdb->get_var($wpdb->prepare("SELECT COUNT(*) FROM {$table} WHERE conversation_id=%d", $conversation_id)));
    $rows = $wpdb->get_results($wpdb->prepare("SELECT * FROM {$table} WHERE conversation_id=%d ORDER BY sent_at ASC, _ID ASC LIMIT %d OFFSET %d", $conversation_id, $per_page, $offset));
    $messages = partexhub_msg_normalize_messages($rows, $user_id);
    $summary = partexhub_msg_conversation_row($conversation, $user_id); $summary['currentUserId'] = $user_id;
    return rest_ensure_response(array('conversation'=>$summary,'messages'=>$messages,'pagination'=>array('page'=>$page,'perPage'=>$per_page,'total'=>$total,'totalPages'=>(int)ceil($total/$per_page))));
}

function partexhub_send_message(WP_REST_Request $request) {
    global $wpdb;
    $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $conversation_id = absint($request['id']);
    $conversation = partexhub_msg_get_owned_conversation($conversation_id, $user_id); if (is_wp_error($conversation)) return $conversation;
    $message = trim(sanitize_textarea_field((string)$request->get_param('message_text')));
    if ($message === '') return new WP_Error('partexhub_empty_message', 'Message cannot be empty.', array('status'=>400));
    if (mb_strlen($message) > 3000) return new WP_Error('partexhub_message_too_long', 'Message cannot exceed 3000 characters.', array('status'=>400));
    $reply_id = absint($request->get_param('reply_to_message_id'));
    $reply = null;
    if ($reply_id) { $reply = $wpdb->get_row($wpdb->prepare("SELECT * FROM " . partexhub_msg_tables()['messages'] . " WHERE _ID=%d", $reply_id)); if (!$reply || absint($reply->conversation_id) !== $conversation_id) return new WP_Error('partexhub_invalid_reply', 'The replied-to message is not part of this conversation.', array('status'=>400)); }
    $rate_key = 'partexhub_msg_rate_' . $user_id;
    if (get_transient($rate_key)) return new WP_Error('partexhub_rate_limit', 'Please wait before sending another message.', array('status'=>429));
    set_transient($rate_key, 1, 1);
    $receiver_id = $user_id === absint($conversation->buyer_id) ? absint($conversation->seller_id) : absint($conversation->buyer_id);
    $now = current_time('mysql', true); $messages = partexhub_msg_tables()['messages']; $conversations = partexhub_msg_tables()['conversations'];
    $wpdb->query('START TRANSACTION');
    $inserted = $wpdb->insert($messages, array('cct_status'=>'publish','cct_author_id'=>$user_id,'cct_created'=>$now,'cct_modified'=>$now,'conversation_id'=>$conversation_id,'sender_id'=>$user_id,'receiver_id'=>$receiver_id,'message_text'=>$message,'is_read'=>0,'sent_at'=>$now,'message_type'=>'text','reply_to_message_id'=>$reply_id), array('%s','%d','%s','%s','%d','%d','%d','%s','%d','%s','%s','%d'));
    if (!$inserted) { $wpdb->query('ROLLBACK'); delete_transient($rate_key); return new WP_Error('partexhub_send_failed', 'Message could not be sent.', array('status'=>500)); }
    $message_id = absint($wpdb->insert_id); $counter = $receiver_id === absint($conversation->buyer_id) ? 'buyer_unread_count' : 'seller_unread_count';
    $updated = $wpdb->query($wpdb->prepare("UPDATE {$conversations} SET last_message=%s,last_message_time=%s,cct_modified=%s,{$counter}=COALESCE({$counter},0)+1 WHERE _ID=%d", $message, $now, $now, $conversation_id));
    if ($updated === false) { $wpdb->query('ROLLBACK'); delete_transient($rate_key); return new WP_Error('partexhub_send_failed', 'Message could not be sent.', array('status'=>500)); }
    $wpdb->query('COMMIT');
    $created = $wpdb->get_row($wpdb->prepare("SELECT * FROM {$messages} WHERE _ID=%d", $message_id));
    return new WP_REST_Response(array('message'=>partexhub_msg_normalize_messages(array($created), $user_id)[0]), 201);
}

function partexhub_message_reaction_context($message_id, $user_id) {
    global $wpdb; $tables = partexhub_msg_tables();
    $message = $wpdb->get_row($wpdb->prepare("SELECT * FROM {$tables['messages']} WHERE _ID=%d", $message_id));
    if (!$message) return new WP_Error('partexhub_message_missing', 'Message not found.', array('status'=>404));
    $conversation = partexhub_msg_get_owned_conversation(absint($message->conversation_id), $user_id); if (is_wp_error($conversation)) return $conversation;
    return $message;
}

function partexhub_message_reactions_for($message, $user_id) { $normalized = partexhub_msg_normalize_messages(array($message), $user_id); return $normalized[0]['reactions']; }

function partexhub_set_message_reaction(WP_REST_Request $request) {
    global $wpdb; $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $message_id = absint($request['id']); $reaction = sanitize_key((string)$request->get_param('reaction')); $allowed = array('like','love','laugh','wow','sad');
    if (!in_array($reaction, $allowed, true)) return new WP_Error('partexhub_invalid_reaction', 'Unsupported reaction.', array('status'=>400));
    $message = partexhub_message_reaction_context($message_id, $user_id); if (is_wp_error($message)) return $message;
    $table = partexhub_msg_tables()['reactions']; $existing_ids = $wpdb->get_col($wpdb->prepare("SELECT _ID FROM {$table} WHERE message_id=%d AND user_id=%d ORDER BY _ID ASC", $message_id, $user_id)); $existing = absint($existing_ids[0] ?? 0); $now = current_time('mysql', true);
    if ($existing) { $ok = $wpdb->update($table, array('reaction'=>$reaction,'cct_modified'=>$now), array('_ID'=>$existing), array('%s','%s'), array('%d')); if (count($existing_ids) > 1) { $duplicates = implode(',', array_map('absint', array_slice($existing_ids, 1))); $wpdb->query("DELETE FROM {$table} WHERE _ID IN ({$duplicates})"); } }
    else $ok = $wpdb->insert($table, array('cct_status'=>'publish','cct_author_id'=>$user_id,'cct_created'=>$now,'cct_modified'=>$now,'message_id'=>$message_id,'user_id'=>$user_id,'reaction'=>$reaction), array('%s','%d','%s','%s','%d','%d','%s'));
    if ($ok === false) return new WP_Error('partexhub_reaction_failed', 'Reaction could not be updated.', array('status'=>500));
    return rest_ensure_response(array('reactions'=>partexhub_message_reactions_for($message, $user_id)));
}

function partexhub_delete_message_reaction(WP_REST_Request $request) {
    global $wpdb; $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $message_id = absint($request['id']); $message = partexhub_message_reaction_context($message_id, $user_id); if (is_wp_error($message)) return $message;
    $wpdb->delete(partexhub_msg_tables()['reactions'], array('message_id'=>$message_id,'user_id'=>$user_id), array('%d','%d'));
    return rest_ensure_response(array('reactions'=>partexhub_message_reactions_for($message, $user_id)));
}

function partexhub_profile_preview(WP_REST_Request $request) {
    $user = get_userdata(absint($request['id'])); if (!$user) return new WP_Error('partexhub_user_missing', 'Profile not found.', array('status'=>404));
    $meta = function($key) use ($user) { return trim((string)get_user_meta($user->ID, $key, true)); }; $phone_setting = strtolower($meta('phone_number_public')); $public_phone = in_array($phone_setting, array('yes','1','true'), true) ? $meta('phone_number') : null; $bio = $meta('bio') ?: $user->description;
    return rest_ensure_response(array('id'=>$user->ID,'displayName'=>$user->display_name,'profilePhotoUrl'=>partexhub_msg_avatar($user->ID),'roleLabel'=>in_array('seller',(array)$user->roles,true)?'Seller':'Marketplace Member','companyName'=>$meta('company_name'),'businessType'=>$meta('business_type'),'city'=>$meta('city'),'country'=>$meta('country'),'bioExcerpt'=>partexhub_msg_excerpt($bio),'phone'=>$public_phone ?: null,'publicProfileUrl'=>'/sellers/'.$user->user_nicename));
}

function partexhub_read_conversation(WP_REST_Request $request) {
    global $wpdb;
    $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $conversation_id = absint($request['id']);
    $conversation = partexhub_msg_get_owned_conversation($conversation_id, $user_id); if (is_wp_error($conversation)) return $conversation;
    $tables = partexhub_msg_tables(); $wpdb->query($wpdb->prepare("UPDATE {$tables['messages']} SET is_read=1,cct_modified=%s WHERE conversation_id=%d AND receiver_id=%d AND is_read=0", current_time('mysql', true), $conversation_id, $user_id));
    $field = $user_id === absint($conversation->buyer_id) ? 'buyer_unread_count' : 'seller_unread_count';
    $wpdb->query($wpdb->prepare("UPDATE {$tables['conversations']} SET {$field}=0,cct_modified=%s WHERE _ID=%d", current_time('mysql', true), $conversation_id));
    return rest_ensure_response(array('success'=>true));
}

function partexhub_message_notifications() {
    global $wpdb;
    $ready = partexhub_msg_tables_ready(); if (is_wp_error($ready)) return $ready;
    $user_id = get_current_user_id(); $table = partexhub_msg_tables()['conversations'];
    $rows = $wpdb->get_results($wpdb->prepare("SELECT buyer_id,buyer_unread_count,seller_unread_count FROM {$table} WHERE buyer_id=%d OR seller_id=%d", $user_id, $user_id));
    $conversation_count=0; $message_count=0; foreach($rows as $row) { $count = $user_id===absint($row->buyer_id) ? absint($row->buyer_unread_count) : absint($row->seller_unread_count); if($count>0) $conversation_count++; $message_count += $count; }
    return rest_ensure_response(array('unreadConversationCount'=>$conversation_count,'unreadMessageCount'=>$message_count));
}

add_action('rest_api_init', function() {
    $permission = 'partexhub_msg_permission';
    register_rest_route('partexhub/v1', '/conversations', array(
        array('methods'=>WP_REST_Server::READABLE,'callback'=>'partexhub_get_conversations','permission_callback'=>$permission),
        array('methods'=>WP_REST_Server::CREATABLE,'callback'=>'partexhub_create_conversation','permission_callback'=>$permission),
    ));
    register_rest_route('partexhub/v1', '/conversations/(?P<id>\d+)', array('methods'=>WP_REST_Server::READABLE,'callback'=>'partexhub_get_conversation','permission_callback'=>$permission));
    register_rest_route('partexhub/v1', '/conversations/(?P<id>\d+)/send', array('methods'=>WP_REST_Server::CREATABLE,'callback'=>'partexhub_send_message','permission_callback'=>$permission));
    register_rest_route('partexhub/v1', '/conversations/(?P<id>\d+)/read', array('methods'=>WP_REST_Server::CREATABLE,'callback'=>'partexhub_read_conversation','permission_callback'=>$permission));
    register_rest_route('partexhub/v1', '/notifications', array('methods'=>WP_REST_Server::READABLE,'callback'=>'partexhub_message_notifications','permission_callback'=>$permission));
    register_rest_route('partexhub/v1', '/messages/(?P<id>\d+)/reaction', array(
        array('methods'=>WP_REST_Server::CREATABLE,'callback'=>'partexhub_set_message_reaction','permission_callback'=>$permission),
        array('methods'=>WP_REST_Server::DELETABLE,'callback'=>'partexhub_delete_message_reaction','permission_callback'=>$permission),
    ));
    register_rest_route('partexhub/v1', '/users/(?P<id>\d+)/profile-preview', array('methods'=>WP_REST_Server::READABLE,'callback'=>'partexhub_profile_preview','permission_callback'=>$permission));
});
